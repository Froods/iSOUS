139 def save_desired_values(self, temp_text, co2_level):
140 if self.client is None:
141 print("Error: Ikke forbundet til nogen port")
142 return
143
144 try:
145 temp_value = int(float(temp_text))
146 except ValueError:
147 temp_value = 255
148
149 # Validerer at ønsket temperatur ligger i det tilladte interval.
150 if temp_value < 10 or (temp_value > 30 and temp_value != 255):
151 print("Fejl: Ugyldigt valg. Temperaturen skal være mellem 0 og 40")
152 return
153
154 co2_value = self.co2_values.get(co2_level)
155 if co2_value is None:
156 print("Fejl: Ugyldigt CO2-niveau")
157 return
158
159 self.client.send_desired_values(temp=temp_value, co2=co2_value)
160 self.log_target_values(temp=temp_value, co2=co2_value)
161 # Når der gemmes ønskede værdier, genstartes automatisk styring.
162 self.set_manual_mode(False)

240 def _remove_old_log_lines(self, now):
241 oldest_allowed = now - timedelta(hours=self.LOG_INTERVAL_HOURS)
242 kept_lines = []
243
244 with self.log_path.open("r", encoding="utf-8") as log_file:
245 for line in log_file:
246 try:
247 line_time = datetime.strptime(line[:19], "%Y-%m-%d %H:%M:%S")
248 except ValueError:
249 kept_lines.append(line)
250 continue
251
252 if line_time >= oldest_allowed:
253 kept_lines.append(line)
254
255 with self.log_path.open("w", encoding="utf-8") as log_file:

274 self.window_open = False
