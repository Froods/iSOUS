302 def __pack_values(self, cmd, par1, par2):
303 # --- Parametre til struct.pack ---
304 # 1. Parameter: Hvordan data skal pakkes
305 # 2. Parameter: Kommando
306 # 3. Parameter: Parameter 1 (til kommando)
